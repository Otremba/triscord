# Google's installed-app token exchange needs the desktop client's secret, which Google does not treat as confidential
# (RFC 8252 §8.5). A distributed build carries it from a local ignored file, so other computers can sign in; source
# control never holds it. Without the file, it still comes from TRISCORD_GOOGLE_CLIENT_SECRET or the settings.

  endif()
  file(READ "${TRISCORD_GOOGLE_CLIENT_SECRET_FILE}" TRISCORD_GOOGLE_CLIENT_SECRET)
  string(STRIP "${TRISCORD_GOOGLE_CLIENT_SECRET}" TRISCORD_GOOGLE_CLIENT_SECRET)
  string(LENGTH "${TRISCORD_GOOGLE_CLIENT_SECRET}" google_secret_length)
  if(google_secret_length LESS 16 OR google_secret_length GREATER 128 OR NOT TRISCORD_GOOGLE_CLIENT_SECRET MATCHES "^[A-Za-z0-9_-]+$")
    message(FATAL_ERROR "The Google client secret must have 16-128 URL-safe characters")
  endif()

  target_link_libraries(triscord-web PRIVATE triscord_screen_call triscord_camera triscord_voice triscord_auth triscord_core Qt6::Gui ole32 shlwapi shell32 dwmapi dxgi)
  add_custom_command(TARGET triscord-web POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${TRISCORD_WEBVIEW2_SDK}/x64/WebView2Loader.dll" "$<TARGET_FILE_DIR:triscord-web>")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/web/dist/index.html")
    # Every build (a POST_BUILD step would only run on relink and leave stale assets): drop and copy again,
    # so files removed from dist (e.g. the local-only fonts) do not linger.
    add_custom_target(triscord-web-assets ALL
      COMMAND ${CMAKE_COMMAND} -E rm -rf "$<TARGET_FILE_DIR:triscord-web>/web"
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/web/dist" "$<TARGET_FILE_DIR:triscord-web>/web"
      VERBATIM)
    add_dependencies(triscord-web-assets triscord-web)
  endif()
  if(TRISCORD_BUILD_DEV_TOOLS)
    target_compile_definitions(triscord-web PRIVATE TRISCORD_WEB_DEVTOOLS=1)
  else()
    set_target_properties(triscord-web PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
endif()
install(TARGETS triscord-native RUNTIME DESTINATION .)
